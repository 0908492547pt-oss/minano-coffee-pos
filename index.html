CREATE TABLE IF NOT EXISTS minano_pos_sync (
    id INT PRIMARY KEY DEFAULT 1,
    payload JSONB NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Mở quyền đọc/ghi API cho ứng dụng POS (Bật RLS & Policy)
ALTER TABLE minano_pos_sync ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public read and write"
ON minano_pos_sync
FOR ALL
USING (true)
WITH CHECK (true);
